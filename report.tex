% Espacement des titres de chapitre (défaut book : 50 pt au-dessus, 40 pt en dessous)
\makeatletter
\newlength{\chapskipabove}\setlength{\chapskipabove}{-40pt}
\newlength{\chapskipbelow}\setlength{\chapskipbelow}{14pt}
\renewcommand{\@makechapterhead}[1]{%
  \vspace*{\chapskipabove}%
  {\parindent\z@\raggedright\normalfont\interlinepenalty\@M
   \ifnum\c@secnumdepth>\m@ne\if@mainmatter
     \huge\bfseries\@chapapp\space\thechapter\quad
   \fi\fi
   \huge\bfseries #1\par\nobreak
   \vskip 4pt\hrule height 0.6pt
   \vskip\chapskipbelow}}
\renewcommand{\@makeschapterhead}[1]{%
  \vspace*{\chapskipabove}%
  {\parindent\z@\raggedright\normalfont\interlinepenalty\@M
   \huge\bfseries #1\par\nobreak
   \vskip 4pt\hrule height 0.6pt
   \vskip\chapskipbelow}}
\makeatother

% Espacement vertical : book/twoside active \flushbottom, qui étire la colle
% élastique (\parskip) pour égaliser la hauteur des pages. Combiné aux figures [H],
% cela creuse de gros trous entre les paragraphes. On laisse plutôt le mou en bas
% de page et on limite l'élasticité de \parskip.
\raggedbottom
\setlength{\parskip}{6pt plus 1pt minus 2pt} % défaut de la classe : \medskipamount

% La classe met -12ex d'espace avant chaque \subsection (~55 pt, 3,5 lignes),
% alors que \section est a -2.5ex et \subsubsection a -3ex : vraisemblablement une
% coquille pour -1.2ex. Ramene a l'echelle des autres niveaux (87 sous-sections).
\makeatletter
\renewcommand\subsection{\@startsection{subsection}{2}{\z@}%
    {-2.5ex\@plus -1ex \@minus -.2ex}%
    {1.0ex \@plus .2ex}%
    {\normalfont\large\bfseries\itshape}}
\makeatother
